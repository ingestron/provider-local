/** Freeze all sixteen sources for current installed source-to-result acceptance.
 * No source reads, production package edits, public tags or sibling runtime imports.
 */
import assert from "node:assert/strict";
import {
  cpSync,
  mkdirSync,
  mkdtempSync,
  readFileSync,
  readdirSync,
  realpathSync,
  writeFileSync,

import { fileURLToPath, pathToFileURL } from "node:url";

import { createHash } from "node:crypto";

const required = (key) => {
  assert.ok(process.env[`INGESTRON_ACCEPTANCE_${key}`], `Set ${key}`);
  return realpathSync(process.env[`INGESTRON_ACCEPTANCE_${key}`]);
};
const owner = process.env.INGESTRON_ACCEPTANCE_SOURCE_OWNER
    ? required("SOURCE_OWNER")
    : resolve(dirname(fileURLToPath(import.meta.url)), ".."),
  cliRoot = required("CLI_PACKAGE"),
  coreRoot = realpathSync(resolve(cliRoot, "../@ingestron/core")),
  cli = join(cliRoot, "build/cli/cli/index.js"),
  providerOwner = required("PROVIDER"),
  output = required("OUTPUT"),
  python = required("PYTHON");
const require = createRequire(join(cliRoot, "package.json")),
  { parse, stringify } = require("yaml"),
  { installPackage } = await import(
    pathToFileURL(join(coreRoot, "dist/sdk/adapter.js")).href
  );
const cliPackage = JSON.parse(readFileSync(join(cliRoot, "package.json"))),
  corePackage = JSON.parse(readFileSync(join(coreRoot, "package.json")));
assert.ok(
  ["0.17.18/0.12.42", "0.17.18/0.12.43", "0.17.20/0.12.47", "0.17.21/0.12.50", "0.17.21/0.12.51", "0.17.21/0.12.53"].includes(
    cliPackage.version + "/" + corePackage.version,
  ),
  "Use an exact qualified private archive pair",
);
assert.equal(cliPackage.dependencies["@ingestron/core"], "0.12.14");
// The private acceptance override is explicit; the public pin stays unchanged.
const sha = (file) =>
  createHash("sha256").update(readFileSync(file)).digest("hex");
const archives = Object.fromEntries(
  ["CORE_ARCHIVE", "CLI_ARCHIVE"].map((key) => {
    const path = required(key);
    return [key, { path, sha256: sha(path) }];
  }),
);
const base = mkdtempSync(join(output, "source-runtime-matrix-")),
  root = join(base, "project"),
  sources = join(base, "source-origin"),
  provider = join(base, "provider-origin"),
  fixtures = join(base, "fixtures"),
  data = join(root, "data");
for (const folder of [root, sources, provider, data])
  mkdirSync(folder, { recursive: true });
const ids = readdirSync(join(owner, "connectors"), { withFileTypes: true })
  .filter((p) => p.isDirectory())
  .map((p) => p.name)
  .sort();
assert.equal(ids.length, 16);
for (const id of ids)
  cpSync(join(owner, "connectors", id), join(sources, "connectors", id), {
    recursive: true,
  });
cpSync(join(providerOwner, "plugin"), join(provider, "plugin"), {
  recursive: true,
  filter: (p) => !p.includes("__pycache__"),
});
cpSync(join(owner, "test/mocks"), join(fixtures, "mocks"), {
  recursive: true,
  filter: (p) => !p.includes("__pycache__"),
});
cpSync(
  join(owner, "test/github_fixture.py"),
  join(fixtures, "github_fixture.py"),
);
const git = (cwd, ...args) =>
  execFileSync("git", args, { cwd, encoding: "utf8", stdio: "pipe" }).trim();
for (const cwd of [sources, provider]) {
  git(cwd, "init", "-q");
  git(cwd, "add", ".");
  git(
    cwd,
    "-c",
    "user.name=Synthetic fixture",
    "-c",
    "user.email=fixture@example.invalid",
    "commit",
    "-qm",
    "Frozen current source runtime matrix",
  );
}
const providerManifest = parse(
  readFileSync(join(provider, "plugin/provider.yaml"), "utf8"),
);
assert.equal(providerManifest.version, "0.4.16");
git(provider, "tag", providerManifest.version);
const providerRef = `ingestron/provider-local/plugin/provider.yaml@${providerManifest.version}`;
installPackage(root, providerRef, { fromGit: provider });

const packages = { local: providerRef },
  connections = {},
  flows = [],
  records = [];
for (const id of ids) {
  const folder = join(sources, "connectors", id),
    manifest = parse(readFileSync(join(folder, "connector.yaml"), "utf8"));
  assert.equal(sha(join(folder, "runtime.json")), manifest.runtime.sha256);
  const runtime = JSON.parse(readFileSync(join(folder, "runtime.json")));
  assert.ok(
    !Object.keys(runtime).some((p) => /fixture|loopback|mock/.test(p)),
    "Fixtures must not ship as runtime assets",
  );
  git(sources, "tag", `${id}-${manifest.version}`);
  const reference = `ingestron/connectors/connectors/${id}/connector.yaml@${manifest.version}`;
  installPackage(root, reference, { fromGit: sources, tagPrefix: `${id}-` });
  packages[id] = reference;
  connections[id] = {
    package: id,
    sourceId: id,
    tenantId: "synthetic",
    settings: cases[id].settings,
  };
  const columns = cases[id].columns ?? ["id", "label"],
    flow = `flow_${id.replaceAll("-", "_")}`;
  flows.push({
    apiVersion: "ingestron.flow/v1",
    id: flow,
    kind: "ingestion",
    provider: "local",
    ingestion: { connection: id, execution: { mode: "local" } },
    tables: {
      records: {
        source: cases[id].source,
        contract: {
          apiVersion: "v3.1.0",
          kind: "DataContract",
          id: `contract_${id}`,
          name: "records",
          version: "1.0.0",
          status: "draft",
          schema: [
            {
              name: "records",
              logicalType: "object",
              physicalType: "table",
              properties: columns.map((name, index) => ({
                name,
                logicalType:
                  index === 0 && !cases[id].stringKey ? "integer" : "string",
                physicalType:
                  index === 0 && !cases[id].stringKey ? "BIGINT" : "STRING",
                required: true,
                ...(index === 0 ? { primaryKey: true } : {}),
              })),
            },
          ],
        },
      },
    },
  });
  records.push({
    id,
    version: manifest.version,
    reference,
    manifestSha256: sha(join(folder, "connector.yaml")),
    runtimeSha256: manifest.runtime.sha256,
    actions: manifest.sourceAccess.actions,
    flow,
    source: cases[id].source,
    scope: cases[id].scope,
    columns,
    rows: cases[id].rows ?? 3,
    secretName: secretName(id),

    environments: {

// Only copy existing qualified cache entries; missing locks are prepared normally.
for (const cache of JSON.parse(
  process.env.INGESTRON_ACCEPTANCE_CACHES ?? "[]",
)) {
  for (const key of readdirSync(cache, { withFileTypes: true })
    .filter((p) => p.isDirectory() && /^[a-f0-9]{64}$/.test(p.name))
    .map((p) => p.name)) {
    for (const area of ["runtime-cache", "runtimes"])
      cpSync(join(cache, key), join(root, ".ingestron", area, key), {
        recursive: true,
      });
  }
}
